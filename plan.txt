================================================================================
1. PROJECT OVERVIEW & BUSINESS LOGIC
================================================================================

SproutBox is a B2B food-tech SaaS platform with 3 user roles:
  - RESTAURANT: Places orders for microgreens, receives deliveries, gives feedback
  - GROWER: Home-based grower who receives tray tasks, grows, uploads progress
  - ADMIN: Platform operator who runs demand engine, allocates tasks, reviews QC

CORE FLOW:
  Restaurant places order
    → Admin demand engine converts kg → trays (with 20-40% buffer)
    → Allocation engine scores growers and distributes tray tasks
    → Growers receive day-by-day instructions and upload milestone images
    → Admin reviews images via digital QC (Pass / Risk / Reject)
    → Growers harvest approved batches and drop off at hub
    → Hub operator does physical QC and aggregates
    → Delivery to restaurant
    → Restaurant gives feedback
    → Feedback updates grower scores → better allocation next cycle

KEY CALCULATIONS:
  Tray yield standard: 1 tray = 200g harvest
  Buffer: Add 25% extra trays to production plan (configurable per crop)
  Grower score = (yieldAccuracy * 0.4) + (qualityPassRate * 0.4) + (timeliness * 0.2)
  Allocation: trays distributed proportional to grower score within same cluster
  Growth cycle: 7-10 days depending on crop type

CROP TYPES SUPPORTED (MVP):
  - Sunflower shoots (9 days)
  - Pea shoots (8 days)
  - Radish (7 days)
  - Wheat grass (10 days)
  - Mustard (7 days)
  - Fenugreek (8 days)

  GROWER
  RESTAURANT

  CONFIRMED
  IN_PRODUCTION

  IN_TRANSIT
  DELIVERED
  CANCELLED

  ASSIGNED
  IN_PROGRESS
  HARVEST_READY
  HARVESTED
  CANCELLED
}

enum BatchStatus {
  SOWING
  GROWING
  PRE_HARVEST
  QC_PENDING
  QC_PASSED
  QC_FAILED
  HARVESTED
  REJECTED

enum QCResult {

  RISK
  REJECT

  FAILED

  FAILED

  id              String    @id @default(cuid())

  phone           String
  address         String
  city            String
  pincode         String

  lng             Float?

  compositeScore  Float     @default(0.5)

  joinedAt        DateTime  @default(now())

  cuisineType      String?

  isActive         Boolean   @default(true)

--- DEMAND ENGINE CALCULATION ---

function calculateProductionPlan(orders: Order[], bufferPercent = 0.25) {
  // Group by crop type
  const grouped = orders.reduce((acc, order) => {
    acc[order.cropType] = (acc[order.cropType] || 0) + order.quantityKg;
    return acc;
  }, {} as Record<string, number>);

  return Object.entries(grouped).map(([cropType, totalKg]) => {
    const baseTrays = Math.ceil((totalKg * 1000) / 200); // 200g per tray
    const bufferTrays = Math.ceil(baseTrays * bufferPercent);
    const totalTrays = baseTrays + bufferTrays;
    const cycleDays = CROP_CYCLE_DAYS[cropType] || 9;
    const sowDate = new Date(); // sow today
    const harvestDate = addDays(sowDate, cycleDays);

    return { cropType, totalKg, baseTrays, bufferTrays, totalTrays, sowDate, harvestDate };
  });
}

--- ALLOCATION ALGORITHM ---

function allocateTrays(growers: Grower[], totalTrays: number, clusterId: string) {
  // Filter to same cluster, active growers with capacity
  const eligible = growers
    .filter(g => g.clusterId === clusterId && g.isActive)
    .sort((a, b) => b.compositeScore - a.compositeScore);

  const totalScore = eligible.reduce((sum, g) => sum + g.compositeScore, 0);

  let assigned = 0;
  const allocations = eligible.map((grower, i) => {
    // Last grower gets remainder
    const trays = i === eligible.length - 1
      ? totalTrays - assigned
      : Math.round((grower.compositeScore / totalScore) * totalTrays);
    assigned += trays;
    return { growerId: grower.id, trayCount: trays };
  });

  return allocations.filter(a => a.trayCount > 0);
}

--- GROWER SCORE UPDATER ---

async function updateGrowerScores(growerId: string) {
  const recentBatches = await prisma.batch.findMany({
    where: { task: { growerId }, status: { in: ['HARVESTED', 'REJECTED'] } },
    include: { task: true },
    orderBy: { harvestedAt: 'desc' },
    take: 20,
  });

  const harvested = recentBatches.filter(b => b.status === 'HARVESTED');

  // Yield accuracy: actual vs expected
  const yieldScore = harvested.length > 0
    ? harvested.reduce((sum, b) => {
        const expected = b.task.trayCount * 200; // grams
        return sum + Math.min((b.actualYieldGrams || 0) / expected, 1.0);
      }, 0) / harvested.length
    : 0.5;

  // Quality: pass rate
  const qualityScore = recentBatches.length > 0
    ? harvested.length / recentBatches.length
    : 0.5;

  // Timeliness: harvested on or before target date
  const timelinessScore = harvested.length > 0
    ? harvested.filter(b =>
        b.harvestedAt && b.harvestedAt <= b.task.harvestDate
      ).length / harvested.length
    : 0.5;

  const compositeScore = (yieldScore * 0.4) + (qualityScore * 0.4) + (timelinessScore * 0.2);

  await prisma.grower.update({
    where: { id: growerId },
    data: { yieldScore, qualityScore, timelinessScore, compositeScore },
  });
}
